const express = require("express"); const cors = require("cors"); const path = require("path"); const app = express(); app.use(express.json()); app.use(cors()); /* * A Vercel executa esta aplicação como função serverless. * O db.json é usado somente como banco inicial. * * IMPORTANTE: * Os dados novos ficam em memória enquanto a instância estiver ativa. * Para persistência definitiva em produção, depois devemos ligar * o projeto a um banco externo (Postgres/Supabase/Neon etc.). */ const DB_FILE = path.join(__dirname, "../backend/db.json"); let db; function criarBancoInicial() { return { usuarios: [], pacientes: [], triagens: [], consultas: [], tv_chamada: null, tv_historico: [] }; } function carregarBanco() { if (db) return db; try { delete require.cache[require.resolve(DB_FILE)]; const inicial = require(DB_FILE); db = { ...criarBancoInicial(), ...inicial }; if (!Array.isArray(db.usuarios)) db.usuarios = []; if (!Array.isArray(db.pacientes)) db.pacientes = []; if (!Array.isArray(db.triagens)) db.triagens = []; if (!Array.isArray(db.consultas)) db.consultas = []; if (!Array.isArray(db.tv_historico)) db.tv_historico = []; if (!("tv_chamada" in db)) db.tv_chamada = null; } catch (erro) { console.error("Erro ao carregar db.json:", erro); db = criarBancoInicial(); } return db; } function salvarBanco(data) { /* * Não usamos fs.writeFileSync aqui. * O filesystem da função serverless não deve ser usado * como banco persistente. */ db = data; } /* ========================================================= LOGIN ========================================================= */ app.post("/login", (req, res) => { try { const usuario = String(req.body?.usuario || "").trim(); const senha = String(req.body?.senha || ""); if (!usuario || !senha) { return res.status(400).json({ sucesso: false, erro: "Usuário e senha são obrigatórios." }); } const banco = carregarBanco(); const user = banco.usuarios.find( u => String(u.usuario || "").trim() === usuario && String(u.senha || "") === senha ); if (!user) { return res.status(401).json({ sucesso: false, erro: "Usuário ou senha inválidos." }); } return res.json({ sucesso: true, usuario: user.usuario, tipo: user.tipo }); } catch (erro) { console.error("Erro no login:", erro); return res.status(500).json({ sucesso: false, erro: "Erro interno no servidor." }); } }); /* ========================================================= ATENDIMENTO ========================================================= */ app.post("/atendimento", (req, res) => { try { const banco = carregarBanco(); const paciente = { id: Date.now(), nome: req.body?.nome || "", cpf: req.body?.cpf || "", tipo: req.body?.tipo || "Particular", status: "triagem", createdAt: new Date().toISOString() }; banco.pacientes.push(paciente); salvarBanco(banco); return res.status(201).json(paciente); } catch (erro) { console.error("Erro no atendimento:", erro); return res.status(500).json({ erro: "Não foi possível cadastrar o paciente." }); } }); /* ========================================================= PACIENTES ========================================================= */ app.get("/pacientes", (req, res) => { try { const banco = carregarBanco(); return res.json(banco.pacientes); } catch (erro) { console.error("Erro ao listar pacientes:", erro); return res.status(500).json({ erro: "Não foi possível carregar os pacientes." }); } }); /* ========================================================= TRIAGEM ========================================================= */ app.post("/triagem", (req, res) => { try { const banco = carregarBanco(); const temperatura = req.body?.temperatura !== undefined && req.body?.temperatura !== "" ? Number(req.body.temperatura) : null; let risco = req.body?.risco; if (temperatura !== null && temperatura >= 39) { risco = "vermelho"; } else if (temperatura !== null && temperatura >= 38) { risco = "amarelo"; } else if (!risco) { risco = "verde"; } const triagem = { id: Date.now(), nome: req.body?.nome || "", sintoma: req.body?.sintoma || req.body?.sintomas || "", temperatura, alergia: req.body?.alergia || "", observacao: req.body?.observacao || "", risco, status: "aguardando_medico", createdAt: new Date().toISOString() }; banco.triagens.push(triagem); salvarBanco(banco); return res.status(201).json(triagem); } catch (erro) { console.error("Erro na triagem:", erro); return res.status(500).json({ erro: "Não foi possível salvar a triagem." }); } }); /* ========================================================= TRIAGENS ========================================================= */ app.get("/triagens", (req, res) => { try { const banco = carregarBanco(); return res.json(banco.triagens); } catch (erro) { console.error("Erro ao listar triagens:", erro); return res.status(500).json({ erro: "Não foi possível carregar as triagens." }); } }); /* ========================================================= TV - CHAMAR PACIENTE ========================================================= */ app.post("/tv/chamar", (req, res) => { try { const banco = carregarBanco(); const chamada = { id: Date.now().toString(), localTipo: req.body?.localTipo || "", localNumero: req.body?.localNumero || "", paciente: req.body?.paciente || "", hora: new Date().toLocaleTimeString("pt-BR", { hour: "2-digit", minute: "2-digit" }) }; banco.tv_chamada = chamada; banco.tv_historico.unshift(chamada); if (banco.tv_historico.length > 5) { banco.tv_historico = banco.tv_historico.slice(0, 5); } salvarBanco(banco); return res.json(chamada); } catch (erro) { console.error("Erro ao chamar paciente:", erro); return res.status(500).json({ erro: "Não foi possível chamar o paciente." }); } }); /* ========================================================= TV - CONSULTAR CHAMADA ========================================================= */ app.get("/tv/chamada", (req, res) => { try { const banco = carregarBanco(); return res.json({ chamada: banco.tv_chamada, historico: banco.tv_historico }); } catch (erro) { console.error("Erro ao consultar TV:", erro); return res.status(500).json({ erro: "Não foi possível carregar a chamada da TV." }); } }); /* ========================================================= LISTA DE MEDICAÇÕES ========================================================= */ app.get("/lista-medicacoes", (req, res) => { return res.json([ "Dipirona", "Paracetamol", "Ibuprofeno", "Amoxicilina", "Azitromicina", "Loratadina", "Omeprazol", "Buscopan", "Dramin", "Soro fisiológico" ]); }); /* ========================================================= CONSULTA ========================================================= */ app.post("/consulta", (req, res) => { try { const banco = carregarBanco(); const consulta = { id: Date.now(), paciente: req.body?.paciente || "", diagnostico: req.body?.diagnostico || "", medicacao: req.body?.medicacao || "", obs: req.body?.obs || "", createdAt: new Date().toISOString() }; banco.consultas.push(consulta); salvarBanco(banco); return res.status(201).json(consulta); } catch (erro) { console.error("Erro na consulta:", erro); return res.status(500).json({ erro: "Não foi possível salvar a consulta." }); } }); /* ========================================================= MEDICAÇÕES / CONSULTAS ========================================================= */ app.get("/medicacoes", (req, res) => { try { const banco = carregarBanco(); return res.json(banco.consultas); } catch (erro) { console.error("Erro ao listar consultas:", erro); return res.status(500).json({ erro: "Não foi possível carregar as consultas." }); } }); /* ========================================================= HEALTH CHECK ========================================================= */ app.get("/api/health", (req, res) => { return res.json({ status: "ok", sistema: "Projeto Sentinela" }); }); /* ========================================================= ERRO 404 ========================================================= */ app.use((req, res) => { res.status(404).json({ erro: "Rota não encontrada." }); }); /* ========================================================= EXPORTAÇÃO PARA VERCEL ========================================================= */ module.exports = app;
